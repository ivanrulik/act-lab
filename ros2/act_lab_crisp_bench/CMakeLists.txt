cmake_minimum_required(VERSION 3.16)
project(act_lab_crisp_bench)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pinocchio REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(crisp_bench src/bench.cpp)
target_compile_features(crisp_bench PRIVATE cxx_std_17)
ament_target_dependencies(crisp_bench controller_interface hardware_interface
  pluginlib rclcpp geometry_msgs)
target_link_libraries(crisp_bench pinocchio::pinocchio nlohmann_json::nlohmann_json)
install(TARGETS crisp_bench DESTINATION lib/${PROJECT_NAME})
ament_package()
